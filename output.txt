func (c *Client) HandleCommand() error {
	// Get kong context
	ctx, err := c.GetContext(os.Args[1:])
	if err != nil {
		return err
	}

	// Handle valid commands
	var res error
	i_struct := &pr.CLI.IpcIdentifier
	ipc := i_struct.IpcIdentifier
	command := ctx.Selected().Name

	switch command {
	// Create
	case pr.CmdCreate:
		i_struct.Create.IpcIdentifier = ipc
		res = cmd.HandleCmdCreate(ctx)

	// List
	case pr.CmdList:
		i_struct.List.IpcIdentifier = ipc
		res = cmd.HandleCmdList(ctx)

	// Produce
	case pr.CmdProduce:
		i_struct.Produce.IpcIdentifier = ipc
		res = cmd.HandleCmdProduce(ctx)

	// Subscribe
	case pr.CmdSubscribe:
		i_struct.Subscribe.IpcIdentifier = ipc
		res = cmd.HandleCmdSubscribe(ctx)

	// Info
	case pr.CmdInfo:
		i_struct.Info.IpcIdentifier = ipc
		res = cmd.HandleCmdInfo(ctx)

	// Handle invalid command
	default:
		res = errors.New(
			"Invalid command. Need to be: create, list, produce, subscribe or info.",
		)
	}

	return res
}

func (c *Client) GetContext(args []string) (*kong.Context, error) {
	// Create kong parser
	parser, err := kong.New(&pr.CLI,
		kong.Name("client"),
		kong.Description("Treenity Message Queue System Client"),
	)

	// Handle initialization error
	if err != nil {
		return nil, err
	}

	// Parse arguments
	ctx, err := parser.Parse(args)
	if err != nil {
		return nil, err
	}

	return ctx, nil
}

func HandleCmdSubscribe(command pr.CmdSubscribeDatas, topics *t.Topics, clients *h.Hashmap) (string, any, error) {
	var client *pr.Client
	var err error

	// Check topic exist
	topic, exist := topics.Get(command.TopicName)
	if !exist {
		return "", "", fmt.Errorf("'%s' topic does not exist", command.TopicName)
	}

	client, err = clients.Get(command.SubscriberName)

	// Check subscriber is valid name
	if !IsValidId(command.SubscriberName) {
		return "", "", fmt.Errorf("invalid subscriber name '%s', must respect this format `^[a-zA-Z0-9_.-]+$`", command.SubscriberName)
	}

	// Check subscriber doesn't exist or is reconnecting
	if err == nil {
		// Check client is connected
		if client.IsActive {
			return "", "", fmt.Errorf("'%s' subscriber is already active", client.Id)
		}

		// Update client datas
		client.IsActive = true
		client.Topic = command.TopicName

		// Update prefix if needed
		if command.Prefix != nil {
			client.PreviousPrefix = client.Prefix
			client.Prefix = *command.Prefix
		}

		// Update offset if needed
		if command.Offset != nil {
			client.Offset = uint32(*command.Offset)
		}

	} else {
		// Handle missing offset option
		offset := uint32(0)
		if command.Offset != nil {
			offset = uint32(*command.Offset)
		}

		// Handle missing prefix option
		prefix := ""
		if command.Prefix != nil {
			prefix = *command.Prefix
		}

		// Create subscriber
		client = &pr.Client{
			Id:       command.SubscriberName,
			Topic:    command.TopicName,
			Offset:   offset,
			Prefix:   prefix,
			IpcPath:  "TODO",
			IsActive: true,
		}

		// Add client to Hashmap
		clients.Append(command.SubscriberName, client)
	}

	// Add client to topic prefix tree
	topic.Tree.Insert(client)

	return "subscribed to " + command.TopicName, "", nil
}

func (s *Server) HandleCommands(command pr.ToServer) (string, any, error) {
	// Define return variables
	var res string
	var datas any
	var err error

	// Handle Command
	switch command.Command {
	case pr.CmdCreate:
		// Cast datas
		command_datas, ok := command.Datas.(pr.CmdCreateDatas)
		if !ok {
			return "", "", errors.New("cast error")
		}

		// Run handler
		res, datas, err = cmd.HandleCmdCreate(command_datas, s.Topics)

	case pr.CmdList:
		// Cast datas
		command_datas, ok := command.Datas.(pr.CmdListDatas)
		if !ok {
			return "", "", errors.New("cast error")
		}

		// Run handler
		res, datas, err = cmd.HandleCmdList(command_datas, s.Topics)

	case pr.CmdInfo:
		// Cast datas
		command_datas, ok := command.Datas.(pr.CmdInfoDatas)
		if !ok {
			return "", "", errors.New("cast error")
		}

		// Run handler
		res, datas, err = cmd.HandleCmdInfo(command_datas, s.Clients)

	case pr.CmdProduce:
		// Cast datas
		command_datas, ok := command.Datas.(pr.CmdProduceDatas)
		if !ok {
			return "", "", errors.New("cast error")
		}

		// Run handler
		res, datas, err = cmd.HandleCmdProduce(command_datas)

	case pr.CmdSubscribe:
		// Cast datas
		command_datas, ok := command.Datas.(pr.CmdSubscribeDatas)

			return "", "", errors.New("cast error")
		}

		// Run handler

	}

import (
	"fmt"
	"os"
	pr "treenity/src/protocol"
)

func (h *Hashmap) Append(key string, value *pr.Client) {
	// Handle Thread safety
	h.mu.Lock()
	defer h.mu.Unlock()

	// Get hash
	hash := CalculateHash(key, len(h.Nodes))

	// Handle first element
	if h.Nodes[hash] == nil {
		h.Nodes[hash] = &Node{Key: key, Value: value, Next: nil}
		h.Length++
		return
	}

	// Append datas at hash's index
	h.Length += h.Nodes[hash].Append(key, value)
}

func (h *Hashmap) Get(key string) (*pr.Client, error) {
	// Handle Thread safety
	h.mu.RLock()
	defer h.mu.RUnlock()

	// Get hash
	hash := CalculateHash(key, len(h.Nodes))

	if h.Nodes[hash] == nil {
		return nil, fmt.Errorf("'%s' key not found", key)
	}

	// Get datas at hash's index
	return h.Nodes[hash].Get(key)
}

func (h *Hashmap) Delete(key string) error {
	// Handle Thread safety
	h.mu.Lock()
	defer h.mu.Unlock()

	// Get hash
	hash := CalculateHash(key, len(h.Nodes))

	if h.Nodes[hash] == nil {
		return fmt.Errorf("'%s' key not found", key)
	}

	// Handle last element
	if h.Nodes[hash].Key == key {
		h.Nodes[hash] = h.Nodes[hash].Next
		h.Length--
		return nil
	}

	// Delete datas at hash's index
	diff, err := h.Nodes[hash].Delete(key)

	h.Length -= diff

	return err
}

func (h *Hashmap) Len() int {
	// Handle Thread safety
	h.mu.RLock()
	defer h.mu.RUnlock()

	return h.Length
}

func (h *Hashmap) Contains(key string) bool {
	// Handle Thread safety
	h.mu.RLock()
	defer h.mu.RUnlock()

	// Get hash
	hash := CalculateHash(key, len(h.Nodes))

	if h.Nodes[hash] == nil {
		return false
	}

	return h.Nodes[hash].Contains(key)
}

func (h *Hashmap) Print() {
	fmt.Fprintf(os.Stderr, "\n\nSize: %d\n", h.Size)
	fmt.Fprintf(os.Stderr, "Length: %d\n", h.Length)

	for deep := range h.Size {
		fmt.Fprintf(os.Stderr, "\nCouche num %d\n", deep)
		curr := h.Nodes[deep]
		for curr != nil {
			fmt.Fprintf(os.Stderr, "Key: %s, Value: %s\n", curr.Key, curr.Value)
			curr = curr.Next
		}
	}
}
package hashmap

import (
	"sync"
	pr "treenity/src/protocol"
)

type Hashmap struct {
	Size   int
	Length int
	Nodes  []*Node
	mu     sync.RWMutex
}

func CreateHashmap(size int) *Hashmap {
	nodes := make([]*Node, size)

	return &Hashmap{
		Nodes:  nodes,
		Size:   size,
		Length: 0,
	}
}

func CalculateHash(key string, length int) int {
	// Cast FNV-1a variables
	fnv_prime := uint64(pr.FnvPrime)
	fnv_offset_basis := uint64(pr.FnvOffsetBasis)

	// Get starting datas
	hash := fnv_offset_basis
	data_bytes := []byte(key)

	// Execute hash algorythm
	for _, data_byte := range data_bytes {
		hash ^= uint64(data_byte)
		hash *= fnv_prime
	}

	// Limit hash to hashmap length
	hash %= uint64(length)

	return int(hash)
}

import (
	"fmt"
	pr "treenity/src/protocol"
)

type Node struct {
	Key   string
	Value *pr.Client
	Next  *Node
}

func (n *Node) Append(key string, value *pr.Client) int {
	if n.Key == key {
		n.Value = value
		return 0
	}

	if n.Next != nil {
		return n.Next.Append(key, value)
	}

	n.Next = &Node{
		Key:   key,
		Value: value,
		Next:  nil,
	}

	return 1
}

func (n *Node) Get(key string) (*pr.Client, error) {
	if n.Key == key {
		return n.Value, nil

	} else if n.Next != nil {
		return n.Next.Get(key)
	}

	return nil, fmt.Errorf("'%s' key not found", key)
}

func (n *Node) Delete(key string) (int, error) {
	if n.Next == nil {
		return 0, fmt.Errorf("'%s' key not found", key)
	}

	if n.Next.Key == key {
		n.Next = n.Next.Next
		return 1, nil
	}

	return n.Next.Delete(key)
}

func (n *Node) Contains(key string) bool {
	if n.Key == key {
		return true
	}

	if n.Next == nil {
		return false
	}

	return n.Next.Contains(key)
}

import (
	"errors"
	"fmt"
	"slices"
	pr "treenity/src/protocol"
)

func (t *Tree) Insert(client *pr.Client) error {
	// Handle Thread safety
	t.mu.Lock()
	defer t.mu.Unlock()

	// Handle nil pointers
	if client == nil {
		return errors.New("Cannot insert nil pointer")
	}

	// Handle update
	if client.PreviousPrefix != client.Prefix {
		t.RemoveClient(client, []rune(client.PreviousPrefix))
	}

	return t.InsertClient(client, []rune(client.Prefix))
}

func (t *Tree) InsertClient(client *pr.Client, prefix []rune) error {
	// Insert client
	if len(prefix) == 0 {
		// Handle duplication
		if slices.Contains(t.Clients, client) {
			return fmt.Errorf("'%s' client is already insert inside Tree", client.Id)
		}

		// Insert client inside Tree
		t.Clients = append(t.Clients, client)

		return nil
	}

	// Create tree if needed
	next_char := prefix[0]
	next_tree, ok := t.Children[next_char]
	if !ok {
		next_tree = CreateTree(next_char)
		t.Children[next_char] = next_tree
	}

	return next_tree.InsertClient(client, prefix[1:])
}

func (t *Tree) Remove(client *pr.Client) error {
	// Handle Thread safety
	t.mu.Lock()
	defer t.mu.Unlock()

	if client == nil {
		return errors.New("Cannot remove nil pointer")
	}

	return t.RemoveClient(client, []rune(client.Prefix))
}

func (t *Tree) RemoveClient(client *pr.Client, prefix []rune) error {
	// Remove client
	if len(prefix) == 0 {
		// Handle missing client
		index := slices.Index(t.Clients, client)
		if index == -1 {
			return fmt.Errorf("'%s' client is not insert inside Tree", client.Id)
		}

		// Remove client from Tree
		t.Clients = append(t.Clients[:index], t.Clients[index+1:]...)

		return nil
	}

	// Check next tree exist
	next_char := prefix[0]
	next_tree, ok := t.Children[next_char]
	if !ok {
		return fmt.Errorf("'%s' client is not insert inside Tree, imcomplete path", client.Id)
	}

	// Remove recursively
	err := next_tree.RemoveClient(client, prefix[1:])
	if err != nil {
		return err
	}

	// Clean empty trees
	if len(next_tree.Clients) == 0 && len(next_tree.Children) == 0 {
		delete(t.Children, next_char)
	}

	return nil
}

func (t *Tree) Filter(key string) []*pr.Client {
	// Handle Thread safety
	t.mu.RLock()
	defer t.mu.RUnlock()

	return t.FilterClient([]rune(key))
}

func (t *Tree) FilterClient(key []rune) []*pr.Client {
	// Get root clients
	filtered_clients := make([]*pr.Client, 0)
	filtered_clients = append(filtered_clients, t.Clients...)

	if len(key) > 0 {
		// Check sub path exist
		sub_tree, exist := t.Children[key[0]]

		// Get sub tree clients
		if exist {
			sub_filtered_clients := sub_tree.FilterClient(key[1:])
			filtered_clients = append(filtered_clients, sub_filtered_clients...)
		}
	}

	return filtered_clients
}

	defer t.mu.RUnlock()

	// Get topic
	topic, exist := t.AllTopics[topic_name]

	return topic, exist
}

func (t *Topics) GetTopicNames() []string {
	// Handle Thread safety
	t.mu.RLock()
	defer t.mu.RUnlock()

	// Get topic
	topics := t.TopicsList

	return topics
}

func TestThreadSafe(t *testing.T) {
	myMap := h.CreateHashmap(1)

	var wg sync.WaitGroup
	workers := 50
	iterations := 100

	// Create a giant monoyy-hash hashmap
	for i := range workers {
		wg.Add(1)
		go func(workerID int) {
			defer wg.Done()
			for j := range iterations {
				key := fmt.Sprintf("key-%d-%d", workerID, j)
				myMap.Append(key, nil)
			}
		}(i)
	}
	wg.Wait()

	// Check init worked
	expectedLen := workers * iterations
	if myMap.Len() != expectedLen {
		t.Errorf("Wanted post appends length %d , got %d", expectedLen, myMap.Len())
	}

	// Reads simultaly to check collisions
	for i := range workers {
		wg.Add(1)
		go func(workerID int) {
			defer wg.Done()
			for j := range iterations {
				key := fmt.Sprintf("key-%d-%d", workerID, j)

				// Read
				_, _ = myMap.Get(key)

				// Check data
				_ = myMap.Contains(key)

				// Delete
				if j%2 == 0 {
					_ = myMap.Delete(key)
				}
			}
		}(i)
	}
	wg.Wait()

	t.Logf("Concurrency tests finished. Final length : %d", myMap.Len())
}

func UtilTestAction(t *testing.T, myMap *h.Hashmap, step sce.HashmapStep) {
	switch step.Action {
	case sce.HashmapActionAppend:
		new_client := &pr.Client{Id: step.Value}
		myMap.Append(step.Key, new_client)

	case sce.HashmapActionGet:
		val, err := myMap.Get(step.Key)
		if (err != nil) != step.WantErr {
			t.Errorf("Get() error = %v, WantErr %v", err, step.WantErr)
		}
		if !step.WantErr && val.Id != step.Want {
			t.Errorf("Get() got = %v, want %v", val, step.Want)
		}

	case sce.HashmapActionDelete:
		err := myMap.Delete(step.Key)
		if (err != nil) != step.WantErr {
			t.Errorf("Delete() error = %v, WantErr %v", err, step.WantErr)
		}

	case sce.HashmapActionContains:
		exists := myMap.Contains(step.Key)
		if exists != step.WantBool {
			t.Errorf("Contains() for key '%s' got = %v, want %v", step.Key, exists, step.WantBool)
		}

	case sce.HashmapActionLen:
		length := myMap.Len()
		if length != step.WantInt {
			t.Errorf("Length() got = %d, want %d", length, step.WantInt)
		}
	}
}

func UtilTestAction(t *testing.T, myTree *p.Tree, step sce.PrefixStep) {
	switch step.Action {
	case sce.PrefixActionInsert:
		// Check prefix update
		if step.CustomPrefix != "" {
			step.Client.Prefix = step.CustomPrefix
		}

		// Insert client
		err := myTree.Insert(step.Client)
		if (err != nil) != step.WantErr {
			t.Errorf("Insert() error = %v, WantErr %v", err, step.WantErr)
		}

	case sce.PrefixActionRemove:
		// Remove client
		err := myTree.Remove(step.Client)
		if (err != nil) != step.WantErr {
			t.Errorf("Remove() error = %v, WantErr %v", err, step.WantErr)
		}

	case sce.PrefixActionFilter:
		// Filter clients
		clients := myTree.Filter(step.Key)
		if !SameClients(clients, step.WantClients) {
			t.Errorf("Filter() with '%s' key, got = %v, want %v", step.Key, clients, step.WantClients)
		}
	}
}

func SameClients(a, b []*pr.Client) bool {
	// Check lengths
	if len(a) != len(b) {
		return false
	}

	// Compare pointers
	for _, client_a := range a {
		if !slices.Contains(b, client_a) {
			return false
		}
	}

	return true
}
